cmake_minimum_required(VERSION 3.24)
project(accelflow_rmsnorm_fp32_dest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${METAL_INTERFACE}")
  message(FATAL_ERROR "Generate METAL_INTERFACE with tools/report/build_rmsnorm_fp32_dest.py")
endif()
include("${METAL_INTERFACE}")
add_executable(rmsnorm_fp32_dest rmsnorm_fp32_dest.cpp)
target_include_directories(rmsnorm_fp32_dest SYSTEM PRIVATE ${METAL_INCLUDES})
target_compile_definitions(rmsnorm_fp32_dest PRIVATE ${METAL_DEFINES}
  ACCELFLOW_KERNEL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/kernels")
target_compile_options(rmsnorm_fp32_dest PRIVATE -O2 -Wall -Wextra)
target_link_libraries(rmsnorm_fp32_dest PRIVATE ${METAL_LINK_LIBRARIES} dl pthread)
set_target_properties(rmsnorm_fp32_dest PROPERTIES BUILD_RPATH "${METAL_LIBRARY_DIRS}")
