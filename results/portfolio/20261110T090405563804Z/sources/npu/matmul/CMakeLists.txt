cmake_minimum_required(VERSION 3.24)
project(accelflow_matmul LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${METAL_INTERFACE}")
  message(FATAL_ERROR "Generate METAL_INTERFACE with tools/report/build_matmul.py")
endif()
include("${METAL_INTERFACE}")
add_executable(matmul matmul.cpp)
target_include_directories(matmul SYSTEM PRIVATE ${METAL_INCLUDES})
target_compile_definitions(matmul PRIVATE ${METAL_DEFINES}
  ACCELFLOW_KERNEL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/kernels")
target_compile_options(matmul PRIVATE -O2 -Wall -Wextra)
target_link_libraries(matmul PRIVATE ${METAL_LINK_LIBRARIES} dl pthread)
set_target_properties(matmul PROPERTIES BUILD_RPATH "${METAL_LIBRARY_DIRS}")
