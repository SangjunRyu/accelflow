cmake_minimum_required(VERSION 3.24)
project(accelflow_reduction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${METAL_INTERFACE}")
  message(FATAL_ERROR "Generate METAL_INTERFACE with tools/report/build_reduction.py")
endif()
include("${METAL_INTERFACE}")
add_executable(reduction reduction.cpp)
target_include_directories(reduction SYSTEM PRIVATE ${METAL_INCLUDES})
target_compile_definitions(reduction PRIVATE ${METAL_DEFINES}
  ACCELFLOW_KERNEL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/kernels")
target_compile_options(reduction PRIVATE -O2 -Wall -Wextra)
target_link_libraries(reduction PRIVATE ${METAL_LINK_LIBRARIES} dl pthread)
set_target_properties(reduction PROPERTIES BUILD_RPATH "${METAL_LIBRARY_DIRS}")
