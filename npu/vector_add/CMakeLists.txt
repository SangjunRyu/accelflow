cmake_minimum_required(VERSION 3.24)
project(accelflow_vector_add LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${METAL_INTERFACE}")
  message(FATAL_ERROR "Generate METAL_INTERFACE with tools/report/build_vector_add.py")
endif()
include("${METAL_INTERFACE}")
foreach(variant riscv tiled)
  add_executable(vector_add_${variant} vector_add_${variant}.cpp)
  target_include_directories(vector_add_${variant} SYSTEM PRIVATE ${METAL_INCLUDES})
  target_compile_definitions(vector_add_${variant} PRIVATE ${METAL_DEFINES}
    ACCELFLOW_KERNEL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/kernels")
  target_compile_options(vector_add_${variant} PRIVATE -O2 -Wall -Wextra)
  target_link_libraries(vector_add_${variant} PRIVATE ${METAL_LINK_LIBRARIES} dl pthread)
  set_target_properties(vector_add_${variant} PROPERTIES BUILD_RPATH "${METAL_LIBRARY_DIRS}")
endforeach()
